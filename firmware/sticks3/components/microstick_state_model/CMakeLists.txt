cmake_minimum_required(VERSION 3.16)

if(COMMAND idf_component_register)
    idf_component_register(SRCS "microstick_state_model.c" INCLUDE_DIRS "include")
    target_compile_options(${COMPONENT_LIB} PRIVATE -Wall -Wextra -Werror)
else()
    project(microstick_state_model LANGUAGES C)
    add_library(microstick_state_model STATIC microstick_state_model.c)
    target_include_directories(microstick_state_model PUBLIC include)
    target_compile_features(microstick_state_model PUBLIC c_std_11)
    target_compile_options(microstick_state_model PRIVATE -Wall -Wextra -Werror)
    enable_testing()
    add_executable(microstick_state_model_test test/microstick_state_model_test.c)
    target_link_libraries(microstick_state_model_test PRIVATE microstick_state_model)
    target_compile_options(microstick_state_model_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME microstick_state_model_test COMMAND microstick_state_model_test)
endif()
