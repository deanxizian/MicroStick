cmake_minimum_required(VERSION 3.16)

if(COMMAND idf_component_register)
    idf_component_register(SRCS "micro_action_tracker.c" INCLUDE_DIRS "include")
    target_compile_options(${COMPONENT_LIB} PRIVATE -Wall -Wextra -Werror)
else()
    project(micro_action_tracker LANGUAGES C)
    add_library(micro_action_tracker STATIC micro_action_tracker.c)
    target_include_directories(micro_action_tracker PUBLIC include)
    target_compile_features(micro_action_tracker PUBLIC c_std_11)
    target_compile_options(micro_action_tracker PRIVATE -Wall -Wextra -Werror)
    enable_testing()
    add_executable(micro_action_tracker_test test/micro_action_tracker_test.c)
    target_link_libraries(micro_action_tracker_test PRIVATE micro_action_tracker)
    target_compile_options(micro_action_tracker_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME micro_action_tracker_test COMMAND micro_action_tracker_test)
endif()
